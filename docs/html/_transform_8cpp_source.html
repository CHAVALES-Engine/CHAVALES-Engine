125void Transform::setParent(Transform* t, bool keepWorldMeasures)
126{
127 if (t == nullptr) return;
128 if (_parent != nullptr && _parent == t) return; // si ya es padre de su hijo
129
130 for (Transform* padre = t; padre != nullptr; padre = padre->getParent())
131 if (padre == this) // de padre en padre buscamos ciclos
132 {
133 Debug::error("Parenting cycle detected, cannot make ", padre, " and ", t, " related.");
134 return; // si lo hay no hacemos nada
135 }
136
137 core::Vector3<> globalPos;
138 core::Quaternion globalRot;
139 core::Vector3<> globalScale;
140
141 if (keepWorldMeasures)
142 {
143 globalPos = getGlobalPosition();
144 globalRot = getGlobalRotation();
145 globalScale = getGlobalScale();
146 }
147
148 // si ya tiene padre
149 if (_parent)
150 _parent->detachChild(this);
151
152 _parent = t;
153 _parent->_children.push_back(this);
154
155 if (keepWorldMeasures)
156 {
157 setGlobalPosition(globalPos);
158 setGlobalRotation(globalRot);
159 setGlobalScale(globalScale);
160 }
161}

165void Transform::detachChild(Transform* c)
166{
167 if (c == nullptr) return;
168
169 std::vector<Transform*>::iterator it = _children.begin();
170 while (it != _children.end())
171 {
172 if (*it == c)
173 {
174 // deshacemos la herencia de medidas
175 core::Vector3<> worldPos = c->getGlobalPosition();
176 core::Quaternion worldRot = c->getGlobalRotation();
177 core::Vector3<> worldScale = c->getGlobalScale();
178
179 c->_parent = nullptr;
180 _children.erase(it);
181
182 c->setLocalPosition(worldPos);
183 c->setLocalRotation(worldRot);
184 c->setLocalScale(worldScale);
185
186 it = _children.end();
187 }
188 ++it;
189 }
190}
