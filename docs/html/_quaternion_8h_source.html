151 inline Quaternion slerp(const Quaternion& a, const Quaternion& b, float t) {
152 float dot = a.dot(b);
153
154 Quaternion b2 = b;
155 if (dot < 0.0f) {
156 dot = -dot;
157 b2._w = -b2._w;
158 b2._x = -b2._x;
159 b2._y = -b2._y;
160 b2._z = -b2._z;
161 }
162
163 float theta_0 = acos(dot);
164 float theta = theta_0 * t;
165
166 float sin_theta = Maths::Sin(theta);
167 float sin_theta_0 = Maths::Sin(theta_0);
168
169 float s0 = std::cos(theta) - dot * sin_theta / sin_theta_0;
170 float s1 = sin_theta / sin_theta_0;
171
172 return Quaternion((a._x * s0) + (b2._x * s1), (a._y * s0) + (b2._y * s1), (a._z * s0) + (b2._z * s1), (a._w * s0) + (b2._w * s1));
173 }

202 inline Vector3<> Euler() {
203 Vector3<> euler;
204 //componentes
205 float sinr_cosp = 2.0f * (_w * _x + _y * _z);
206 float cosr_cosp = 1.0f - 2.0f * (_x * _x + _y * _y);
207 euler.setX(std::atan2(sinr_cosp, cosr_cosp) * 180.0f / M_PI);
208
209 float sinp = 2.0f * (_w * _y - _z * _x);
210 if (std::fabs(sinp) >= 1.0f)
211 euler.setY(std::copysign(90.0f, sinp));
212 else
213 euler.setY(std::asin(sinp) * 180.0f / M_PI);
214
215 float siny_cosp = 2.0f * (_w * _z + _x * _y);
216 float cosy_cosp = 1.0f - 2.0f * (_y * _y + _z * _z);
217 euler.setZ(std::atan2(siny_cosp, cosy_cosp) * 180.0f / M_PI);
218
219 return euler;
220 }
