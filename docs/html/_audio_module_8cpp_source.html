68{
69 auto itSoundFound = _soundMap.find(id);

111int AudioModule::playSound(std::string id, const core::Vector3<> vec3, float soundVolume, int looping)
112{
113 int nextChID = _nextChannelID++;
114 auto itSoundFound = _soundMap.find(id);
115 if (itSoundFound == _soundMap.end())
116 {
117 Debug::error("Sound not found: Couldn't find sound, there isn't a loaded sound with this id: " + id);
118 return -1;
119 }
120 FMOD::Channel* channel = nullptr;
121 _system->playSound(itSoundFound->second, nullptr, true, &channel);
122 //If channel has been correctly created, then registers it
123 if (channel) {
124 FMOD_VECTOR pos = { vec3.getX(),vec3.getY(),vec3.getZ() };
125 channel->set3DAttributes(&pos, nullptr);
126 channel->setVolume(soundVolume);
127 channel->setPaused(false);
128 channel->setLoopCount(looping);
129 _channelSound[nextChID] = channel;
130 }
131 return nextChID;
132}
