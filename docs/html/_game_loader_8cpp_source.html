17void GameLoader::parseObject(const sol::object& obj, const std::string& clave, Properties& props)
18{
19 switch (obj.get_type())
20 {
21 case sol::type::number:
22 {
23 if (obj.is<int>())
24 {
25 props[clave] = obj.as<int>();
26 }
27 else
28 {
29 props[clave] = obj.as<float>();
30 }
31 break;
32 }
33 case sol::type::boolean:
34 {
35 props[clave] = obj.as<bool>();
36 break;
37 }
38 case sol::type::string:
39 {
40 props[clave] = obj.as<std::string>();
41 break;
42 }
43 case sol::type::userdata:
44 {
45 if (obj.is<core::Vector2<>>())
46 {
47 Debug::out("GAMELOADER: vector2");
48 props[clave] = obj.as<core::Vector2<>>();
49 }
50 else if (obj.is<core::Vector3<>>())
51 {
52 Debug::out("GAMELOADER: vector3");
53 props[clave] = obj.as<core::Vector3<>>();
54 }
55 else if (obj.is<core::Vector4<>>())
56 {
57 Debug::out("GAMELOADER: vector4");
58 props[clave] = obj.as<core::Vector4<>>();
59 }
60 else if (obj.is<core::Quaternion<>>())
61 {
62 Debug::out("GAMELOADER: quaternion");
63 props[clave] = obj.as<core::Quaternion<>>();
64 }
65 else if (obj.is<core::Color>())
66 {
67 Debug::out("GAMELOADER: color");
68 props[clave] = obj.as<core::Color>();
69 }
70 else
71 {
72 Debug::error("GAMELOADER: El tipo del parametro de ", clave, " no esta definido.");
73 }
74 break;
75 }
76 default:
77 {
78 Debug::error("GAMELOADER: El tipo del parametro de ", clave, " no es valido.");
79 break;
80 }
81 }
82}

84void GameLoader::parseComponent(core::Entity* e, std::pair<sol::object, sol::object>& componenteObj)
85{
86 // encontramos su nombre e intentamos instanciar un componente con una clase con tal nombre
87 std::string componenteName = componenteObj.first.as<std::string>();
88 std::shared_ptr<core::Component> component = ComponentRegister::instance().create(componenteName);
89
90 if (component != nullptr)
91 {
92 // si este tipo de componente estaba registrado y se ha creado
93 Properties properties;
94
95 sol::table propertiesTable = componenteObj.second;
96 for (auto& p : propertiesTable)
97 {
98 // nombre del parametro
99 std::string nombreParametro = p.first.as<std::string>();
100
101 // declarar variant
102 auto objetoParametro = p.second;
103
104 // traducir objeto a propiedad
105 parseObject(objetoParametro, nombreParametro, properties);
106 }
107
108 // --- a este nivel va el init:
109 // inicializacion de los parametros de un componente a traves de los datos de lua
110 component->init(properties);
111
112 // --- mete el componente a la entidad creada
113 e->addComponent(std::move(component));
114 Debug::out("GAMELOADER: Componente ", componenteName, " cargado para la entidad ", e->getName(), ".");
115 }
116 else
117 {
118 // si no se ha conseguido crear el componente porque no estaba bien registrado
119 Debug::warning("GAMELOADER: Componente ", componenteName, " no registrado.");
120 }
121}

194void GameLoader::loadLua(

196 const sceneName& n,
197 const std::string& p)
198{
199 sol::state lua;
200 lua.open_libraries(sol::lib::base);
201 std::string path = p + n + ".lua";
202
203 defineUserTypes(lua);
204
205 try
206 {
207 // intenta leer archivo
208 lua.safe_script_file(path);
209 }
210 catch (const sol::error& e)
211 {
212 // si no lo consigue saca error
213 Debug::error("GAMELOADER: Error cargando escena: ", path);
214 Debug::error("Lua exception: ", e.what());
215 s = nullptr;
216 return;
217 }
218
219 // --- lectura lua
220 sol::table scene = lua["scene"];
221 Debug::out("GAMELOADER: Escena ", n, " cargada.");
222
223 for (auto& entidadObj : scene)
224 {
225 // --- para cada entidad leida
226 core::Entity* e = new core::Entity();

228 parseEntity(e, entidadObj);

232 //for (auto& c : e->getComponents())

234 // //c->awake();
235 //}
236
237 // --- mete la entidad en la escena
238 s->addEntity(e);
239 }
240
241 // --- a este nivel se llama al ready:
242 // garantizamos que en el ready el resto de entidades y sus componentes estan inicializados
243 s->ready();

275std::string GameLoader::findSceneFile(const std::string& sceneName, const std::string& root)
276{
277 std::string target = sceneName + ".lua";
278
279 for (const auto& entry : fs::recursive_directory_iterator(root))
280 {
281 if (!entry.is_regular_file())
282 continue;
283
284 if (entry.path().filename() == target)
285 {
286 return entry.path().parent_path().string();
287 }
288 }
289
290 return "";
291}
