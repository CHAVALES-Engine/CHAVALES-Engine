18bool ComponentDLLLoader::load(const std::string& path)
19{
20 // Comprobamos duplicados
21 for (const auto& l : _libraries) {

28 HMODULE library = nullptr; // Direccion base de un modulo en memoria
29 // Windows busca una dll y la carga en la memoria del programa
30 if ((library = LoadLibraryA(path.c_str())) == nullptr) {
31 Debug::error("LoadLibrary failed: ", path, " err=", GetLastError());
32 return false;
33 }
34 // TODO: register components
35 // Obtenemos la direccion de memoria de la funcion exportada "getPluginComponents"
36 GetComponentsFn getComponents = (GetComponentsFn)GetProcAddress(library, "getPluginComponents");
37 // Si no se ha devuelto nada lanzamos error y salimos
38 if (!getComponents) {
39 Debug::error("The export components function \"getPluginComponents not\" found in ", path);
40 FreeLibrary(library);
41 return false;
42 }
43 // Cogemos los componentDescriptor de todos los componentes en la dll
44 size_t count = 0;
45 const core::ComponentDescriptor* descriptors = getComponents(count);
46 // Registramos los componentes cargados en el registro del engine
47 for (size_t i = 0; i < count; ++i) {
48 ComponentRegister::instance().registComponent(
49 descriptors[i].name,
50 descriptors[i].ComponentConstructor
51 );
52 }
53
54 // Mete la libreria a un vector
55 _libraries.push_back({ library, path });
56
57 return true;
